cmake_minimum_required(VERSION 3.28)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>
int main(){ std::expected<int,int> x=1; return *x-1; }" GRAPHLAB_HAS_EXPECTED)
if(NOT GRAPHLAB_HAS_EXPECTED)
  message(FATAL_ERROR "C++23 std::expected is required; select a compatible compiler AND standard library")
endif()
