cmake_minimum_required(VERSION 3.28)
project(Graphlab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>
int main(){ std::expected<int,int> x=1; return *x-1; }" GRAPHLAB_HAS_EXPECTED)
if(NOT GRAPHLAB_HAS_EXPECTED)
  message(FATAL_ERROR "C++23 std::expected is required; select a compatible compiler AND standard library")
endif()
include(CTest)
add_subdirectory(packages/lab-support)
add_library(graphlab_planner cpp/planner.cpp)
target_link_libraries(graphlab_planner PUBLIC LabSupport::contracts)
target_include_directories(graphlab_planner PUBLIC include)
target_compile_options(graphlab_planner PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lab cpp/main.cpp)
target_link_libraries(lab PRIVATE graphlab_planner)
target_compile_definitions(lab PRIVATE GRAPHLAB_VERSION="${PROJECT_VERSION}" GRAPHLAB_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
install(TARGETS lab RUNTIME DESTINATION bin)
find_package(Boost 1.92.0 EXACT REQUIRED CONFIG)
find_package(Threads REQUIRED)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(graphlab_catalog cpp/console/inventory.cpp)
target_include_directories(graphlab_catalog PUBLIC include)
target_link_libraries(graphlab_catalog PUBLIC LabSupport::contracts Boost::headers PRIVATE OpenSSL::Crypto)
add_library(graphlab_console cpp/console/rpc.cpp cpp/console/http.cpp)
target_include_directories(graphlab_console PUBLIC include)
target_link_libraries(graphlab_console PUBLIC graphlab_catalog LabSupport::contracts Boost::headers Threads::Threads PRIVATE OpenSSL::Crypto)
target_compile_options(graphlab_console PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lab-agent cpp/console/agent_main.cpp)
add_executable(lab-api cpp/console/api_main.cpp)
target_link_libraries(lab-agent PRIVATE graphlab_console)
target_link_libraries(lab-api PRIVATE graphlab_console)
target_link_libraries(lab PRIVATE graphlab_console)
install(TARGETS lab-agent lab-api RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(m1_tests tests/console/m1.cpp)
  target_link_libraries(m1_tests PRIVATE graphlab_console)
  add_test(NAME m1_acceptance COMMAND m1_tests ${CMAKE_SOURCE_DIR} $<TARGET_FILE:lab-agent> $<TARGET_FILE:lab-api>)
  set_tests_properties(m1_acceptance PROPERTIES TIMEOUT 45)
  add_executable(m0_tests tests/m0.cpp)
  target_link_libraries(m0_tests PRIVATE graphlab_planner)
  add_test(NAME m0_acceptance COMMAND m0_tests)
  add_test(NAME cli_validate COMMAND lab validate ${CMAKE_SOURCE_DIR}/topologies/triangle.yaml --lock ${CMAKE_SOURCE_DIR}/topologies/artifacts.lock.json)
  add_test(NAME cli_plan COMMAND lab plan ${CMAKE_SOURCE_DIR}/topologies/triangle.yaml --lock ${CMAKE_SOURCE_DIR}/topologies/artifacts.lock.json)
  add_test(NAME cli_preflight COMMAND lab preflight)
  add_test(NAME cli_side_effects COMMAND ${CMAKE_COMMAND} -DLAB=$<TARGET_FILE:lab> -DSOURCE=${CMAKE_SOURCE_DIR} -DWORK=${CMAKE_BINARY_DIR}/readonly-test -P ${CMAKE_SOURCE_DIR}/tests/readonly.cmake)
  add_test(NAME installed_package COMMAND ${CMAKE_COMMAND} -DBUILD=${CMAKE_BINARY_DIR} -DSOURCE=${CMAKE_SOURCE_DIR} -DCXX=${CMAKE_CXX_COMPILER} "-DGENERATOR=${CMAKE_GENERATOR}" -P ${CMAKE_SOURCE_DIR}/tests/package.cmake)
endif()
find_package(SQLite3 3.24 REQUIRED)
if(NOT TARGET SQLite3::SQLite3)
  add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
endif()
add_library(graphlab_runtime cpp/runtime/process.cpp cpp/runtime/engine.cpp cpp/runtime/linux.cpp)
target_include_directories(graphlab_runtime PUBLIC include)
target_link_libraries(graphlab_runtime PUBLIC graphlab_catalog LabSupport::contracts Boost::headers Threads::Threads SQLite3::SQLite3)
target_compile_options(graphlab_runtime PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(graphlab_console PUBLIC graphlab_runtime)
if(BUILD_TESTING)
  add_executable(m2_tests tests/runtime/m2.cpp)
  target_link_libraries(m2_tests PRIVATE graphlab_console)
  add_dependencies(m2_tests lab)
  add_test(NAME m2_journal COMMAND m2_tests ${CMAKE_SOURCE_DIR})
  set_tests_properties(m2_journal PROPERTIES TIMEOUT 60)
endif()
if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_executable(m2_linux tests/runtime/linux_integration.cpp)
  target_link_libraries(m2_linux PRIVATE graphlab_console)
  # Explicit invocation only: creates owned Docker/OVS resources and a sentinel.
endif()
