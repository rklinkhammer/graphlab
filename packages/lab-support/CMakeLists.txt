cmake_minimum_required(VERSION 3.28)
project(LabSupport VERSION 1.0.0 LANGUAGES CXX)
include(FetchContent)
# yaml-cpp 0.8.0 predates CMake 4's removal of pre-3.5 compatibility.
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
set(JSON_Install ON CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL ON CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(FETCHCONTENT_TRY_FIND_PACKAGE_MODE NEVER)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" dependency_lock)
foreach(dependency IN ITEMS nlohmann_json yaml-cpp)
  string(JSON dependency_url GET "${dependency_lock}" "${dependency}" url)
  string(JSON dependency_hash GET "${dependency_lock}" "${dependency}" sha256)
  FetchContent_Declare(${dependency} URL "${dependency_url}" URL_HASH "SHA256=${dependency_hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endforeach()
FetchContent_MakeAvailable(nlohmann_json yaml-cpp)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(lab_contracts STATIC cpp/document.cpp cpp/contracts.cpp)
add_library(LabSupport::contracts ALIAS lab_contracts)
set_target_properties(lab_contracts PROPERTIES EXPORT_NAME contracts)
target_compile_features(lab_contracts PUBLIC cxx_std_23)
target_include_directories(lab_contracts PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(lab_contracts PUBLIC nlohmann_json::nlohmann_json PRIVATE yaml-cpp::yaml-cpp OpenSSL::Crypto)
target_compile_options(lab_contracts PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS lab_contracts EXPORT LabSupportTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT LabSupportTargets NAMESPACE LabSupport:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LabSupport)
configure_package_config_file(cmake/LabSupportConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/LabSupportConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LabSupport)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/LabSupportConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LabSupportConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/LabSupportConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LabSupport)
add_library(lab_lifecycle STATIC cpp/lifecycle.cpp)
add_library(LabSupport::lifecycle ALIAS lab_lifecycle)
set_target_properties(lab_lifecycle PROPERTIES EXPORT_NAME lifecycle)
target_compile_features(lab_lifecycle PUBLIC cxx_std_23)
target_include_directories(lab_lifecycle PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(lab_lifecycle PRIVATE nlohmann_json::nlohmann_json)
install(TARGETS lab_lifecycle EXPORT LabSupportTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
